cmake_minimum_required(VERSION 3.20)
project(mimo-talk VERSION 0.2.2 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LVGL_SOURCE_DIR "" CACHE PATH "Cached LVGL 9.5 source")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE FILEPATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
if(LVGL_SOURCE_DIR)
    add_subdirectory("${LVGL_SOURCE_DIR}" lvgl EXCLUDE_FROM_ALL)
    set(lvgl_license "${LVGL_SOURCE_DIR}/LICENCE.txt")
else()
    include(FetchContent)
    FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.5.0 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(lvgl)
    set(lvgl_license "${lvgl_SOURCE_DIR}/LICENCE.txt")
endif()
add_executable(mimo-talk lcd.cpp)
target_link_libraries(mimo-talk PRIVATE lvgl::lvgl m)
target_compile_options(mimo-talk PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mimo-talk RUNTIME DESTINATION bin)
install(FILES mimo.py DESTINATION lib/mimo-talk)
install(FILES mimo-talk.desktop DESTINATION share/APPLaunch/applications)
install(FILES mimo.png DESTINATION share/APPLaunch/share/images RENAME mimo-talk.png)
install(FILES README.md LICENSE config.example.json THIRD_PARTY_NOTICES.md "${lvgl_license}" DESTINATION share/doc/mimo-talk)
option(MIMO_UI_TEST "Build a headless UI render and input test" OFF)
if(MIMO_UI_TEST)
    add_executable(mimo-ui-test tests/ui_test.cpp)
    target_link_libraries(mimo-ui-test PRIVATE lvgl::lvgl m)
    target_compile_options(mimo-ui-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
